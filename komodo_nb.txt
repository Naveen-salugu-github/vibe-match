## External Processing of enrollment data 

============================================================
CELL 1 | markdown
============================================================
### MX **and** RX enrollment

============================================================
CELL 2 | code
============================================================
%sql

CREATE OR REPLACE TABLE us_commercial_catalog_prod.lgu_eplontersen.pt_rx_or_mx_enrt_komodo_0_am AS 
  SELECT pat_id, 
         to_date(eligy_strt_dt) as eligy_strt_dt,
         to_date(eligy_end_dt)  as eligy_end_dt, 
         
         lag(to_date(eligy_end_dt), 1) OVER (PARTITION BY pat_id ORDER BY eligy_strt_dt, eligy_end_dt) as prev_eligy_end_dt, 
         datediff(eligy_strt_dt, prev_eligy_end_dt) as gap,

        case
          when prev_eligy_end_dt is null then 0
          when datediff(eligy_strt_dt, prev_eligy_end_dt) <= 45 then 0
          else 1
        end as qualified_gap_flag        
  FROM us_commercial_catalog_prod.lg_claim.d_claim_pat_enrlt_komodo_attr_v
  where mdcl_claim_cls = True and phcy_claim_cls = True 


============================================================
CELL 3 | code
============================================================
%sql
CREATE OR REPLACE TABLE us_commercial_catalog_prod.lgu_eplontersen.pt_rx_or_mx_enrt_komodo_1_am AS 
  select *,
         sum(qualified_gap_flag) over (partition by pat_id order by eligy_strt_dt, eligy_end_dt rows between unbounded preceding and current row) as enrollment_episode
  from us_commercial_catalog_prod.lgu_eplontersen.pt_rx_or_mx_enrt_komodo_0_am

============================================================
CELL 4 | code
============================================================
%sql
CREATE OR REPLACE TABLE us_commercial_catalog_prod.lgu_eplontersen.pt_rx_or_mx_enrt_komodo_am AS 
   select distinct pat_id, 
          min(eligy_strt_dt) over (partition by pat_id, enrollment_episode) as overall_eligy_strt_dt,
          max(eligy_end_dt)  over (partition by pat_id, enrollment_episode) as overall_eligy_end_dt
  from us_commercial_catalog_prod.lgu_eplontersen.pt_rx_or_mx_enrt_komodo_1_am
