#Treatment data from claims
rx_base=spark.sql(""" select * from us_commercial_catalog_prod.lgu_eplontersen.komodo_raw_treatments_ATTR
where claim_dspn in ("PAID") and claim_sta in ("FINAL","STANDALONE") """)
rx_base.createOrReplaceTempView("rx_base")

rx_base = rx_base.withColumn('prod_desc_lvl3',coalesce(col('prod_desc_lvl3'),when(col('proc_desc_lvl3').contains('AMVUTTRA'),lit('AMVUTTRA FAMILY')).otherwise(when(col('proc_desc_lvl3').contains('ONPATTRO'),lit('ONPATTRO FAMILY')))))

rx_min_max_dates = rx_base.groupBy('pat_id','prod_desc_lvl3')\
    .agg(min_('srvc_dt').alias('first_rx_date'),max_('srvc_dt').alias('recent_rx_date'))

#Creating the sequence of Brands the HCP has prescribed
w_ordered = Window.partitionBy('pat_id').orderBy('first_rx_date')
ordered_set = rx_min_max_dates.withColumn("rank", row_number().over(w_ordered))

ordered_set = ordered_set.withColumn('brand_with_order',struct('rank','prod_desc_lvl3'))
new_pats = ordered_set.groupBy("pat_id") \
     .agg(sort_array(collect_list("brand_with_order")).alias("sorted_brands"))
     
pats_rx_set = new_pats.withColumn("rx_treatment_array", expr("transform(sorted_brands, x -> x.prod_desc_lvl3)")).select("pat_id", "rx_treatment_array")

# Creating the specific dates by brand
stage1_rx = rx_min_max_dates.groupBy('pat_id').pivot('prod_desc_lvl3').agg(min_('first_rx_date').alias('first_rx_date'),min_('recent_rx_date').alias('recent_rx_date'))

stage1_rx = stage1_rx.select([col(x).alias(x.replace(' ', '_')) for x in stage1_rx.columns])

## Creating First Silencer Date and latest silencer date
stage1_rx = stage1_rx.withColumn('SILENCER_first_rx_date',least(stage1_rx["AMVUTTRA_FAMILY_first_rx_date"], stage1_rx["ONPATTRO_FAMILY_first_rx_date"], stage1_rx["WAINUA_FAMILY_first_rx_date"])).withColumn('SILENCER_recent_rx_date',greatest(stage1_rx["AMVUTTRA_FAMILY_recent_rx_date"], stage1_rx["ONPATTRO_FAMILY_recent_rx_date"], stage1_rx["WAINUA_FAMILY_recent_rx_date"]))

## Creating First Stabilizer Date and latest stabilizer date
stage1_rx = stage1_rx.withColumn('STABILIZER_first_rx_date',least(stage1_rx["ATTRUBY_FAMILY_first_rx_date"], stage1_rx["VYNDAMAX_FAMILY_first_rx_date"]))\
    .withColumn('STABILIZER_recent_rx_date',greatest(stage1_rx["ATTRUBY_FAMILY_recent_rx_date"], stage1_rx["VYNDAMAX_FAMILY_recent_rx_date"]))

## Creating Treatment flag - Whether patient is treated with any of the 5 drugs
stage1_rx = stage1_rx.withColumn('rx_treatment_flag',when(
        (stage1_rx['SILENCER_first_rx_date'].isNotNull()) | 
        (stage1_rx['STABILIZER_first_rx_date'].isNotNull()), 
        'Yes').otherwise('No'))

## Creating Silencer only flag
stage1_rx = stage1_rx.withColumn(
    'rx_silencer_only_flag',when((stage1_rx['SILENCER_first_rx_date'].isNotNull()) & 
        (stage1_rx['STABILIZER_first_rx_date'].isNull()),'Yes').otherwise('No')
)

## Creating Stabilizer only flag
stage1_rx = stage1_rx.withColumn(
    'rx_stablizer_only_flag',when((stage1_rx['STABILIZER_first_rx_date'].isNotNull()) & 
        (stage1_rx['SILENCER_first_rx_date'].isNull()),'Yes').otherwise('No')
)

## Creating First Treatment Date - Minimum of Silencer and Stabilizer
stage1_rx = stage1_rx.withColumn('first_treatment_rx_date',least(stage1_rx["SILENCER_first_rx_date"], stage1_rx["STABILIZER_first_rx_date"]))


## Creating First Treatment Column
stage1_rx = stage1_rx.withColumn(
    'first_treatment_rx',
    when(
        (stage1_rx['rx_treatment_flag'] == 'Yes') & 
        (coalesce(date_add(stage1_rx['SILENCER_first_rx_date'], within_same_period_threshold), to_date(lit('2050-01-01'))) < coalesce(stage1_rx['STABILIZER_first_rx_date'], to_date(lit('2050-01-01')))),
        'Silencer'
    )
    .when(
        (stage1_rx['rx_treatment_flag'] == 'Yes') & 
        (coalesce(stage1_rx['SILENCER_first_rx_date'], to_date(lit('2050-01-01'))) > coalesce(date_add(stage1_rx['STABILIZER_first_rx_date'],within_same_period_threshold), to_date(lit('2050-01-01')))),
        'Stabilizer'
    )
    .when(
        (stage1_rx['rx_treatment_flag'] == 'Yes') & 
        (abs_(datediff(stage1_rx['SILENCER_first_rx_date'], stage1_rx['STABILIZER_first_rx_date']))<=within_same_period_threshold),'Combo'
    )
    
    .otherwise('untreated')
)

## Creating Day Difference between First Silencer and First Stabilizer treatment
stage1_rx = stage1_rx.withColumn(
    'rx_time_btwn_silencer_stabilizer',
    datediff(stage1_rx['SILENCER_first_rx_date'], stage1_rx['STABILIZER_first_rx_date'])
)
stage1_rx = stage1_rx.join(pats_rx_set.select(col('pat_id'),col('rx_treatment_array')), on='pat_id', how='left')
stage1_rx.createOrReplaceTempView("stage1_rx")

# Creating flag for patients with closed claims based on the following rule - 
#Patients with >80% of the days between their first ATTR diagnosis and their latest first therapy (silencer or stabilizer) in closed claims 
try:
       del stage2_final_view,overlap_calculation,pat_enrollment_overlap
except NameError:
       print('No variables to delete.')

overlap_threshold_for_enrollment = 0.8
and_or_cols = [data_for_closed_claims.pandas_api().columns[data_for_closed_claims.pandas_api().columns.str.contains('and')],data_for_closed_claims.pandas_api().columns[data_for_closed_claims.pandas_api().columns.str.contains('or')]]

for cols in and_or_cols:
       #Calculating the overlap between patient history and enrollment
       req_cols = data_for_closed_claims.select('pat_id','SILENCER_first_rx_date','STABILIZER_first_rx_date','Dx_Date_of_First_ATTR',*cols).distinct()
       overlap_calculation = req_cols.withColumn("overlap_start", greatest(col("Dx_Date_of_First_ATTR"), coalesce(col(cols[0]),to_date(lit('2050-01-01'))))) \
              .withColumn("overlap_end", least(greatest(col("SILENCER_first_rx_date"),col('STABILIZER_first_rx_date')), coalesce(col(cols[1]),to_date(lit('2050-01-01'))))) \
              .withColumn("overlap_days",
                     when(col("overlap_end") >= col("overlap_start"),
                            datediff(col("overlap_end"), col("overlap_start")) + 1).otherwise(0)) \
              .withColumn("range_1_days", datediff(greatest(col("SILENCER_first_rx_date"),col('STABILIZER_first_rx_date')), col("Dx_Date_of_First_ATTR")) + 1) \
              .withColumn("overlap_percentage", col("overlap_days") / col("range_1_days"))

       #Calculating the total Overlap percentage for each patient in the data
       pat_enrollment_overlap = overlap_calculation.groupBy("pat_id").agg(sum_('overlap_percentage').alias(f"pat_enrollment_overlap{cols[0][-3:]}"))
       try:
             stage2_final_view

              #Adding flag to the data to signify patients that would be considered "Enrolled" based on the above criteria

              .withColumn(f'Enrolled_{cols[0][-3:]}',when(col(f'pat_enrollment_overlap{cols[0][-3:]}') >= lit(overlap_threshold_for_enrollment),'Yes').otherwise('No'))
       else:

date_analysis_cols = ['Dx_Date_of_First_ATTR_CM',

col_names = ['CM_Dx','Tx','PN_Dx','ATTR_Dx','CM_Dx_90','Tx_90','PN_Dx_90','ATTR_Dx_90']
def safe_sort_dates(*dates):
    # Zip and sort, treating None as far-future for ordering purposes
    date_map = zip(col_names, dates)
    sorted_dates = sorted(date_map, key=lambda x: (x[1] is None, x[1]))
    return [col_ for col_, _ in sorted_dates]

sort_udf = udf(safe_sort_dates, ArrayType(StringType()))


stage2_final_view = stage2_final_view.withColumn("Patient_Journey", sort_udf(*date_analysis_cols))

silencer_rx_dt_cols = stage2_final_view.pandas_api().columns[stage2_final_view.pandas_api().columns.str.contains('AMV|WAI|ONP')& stage2_final_view.pandas_api().columns.str.contains('first')]
wai_amv_rx_dt_cols = stage2_final_view.pandas_api().columns[stage2_final_view.pandas_api().columns.str.contains('AMV|WAI')& stage2_final_view.pandas_api().columns.str.contains('first')]
stabilizer_rx_dt_cols = stage2_final_view.pandas_api().columns[stage2_final_view.pandas_api().columns.str.contains('VYN|ATTRU')& stage2_final_view.pandas_api().columns.str.contains('first')]

# silencer_first_rx_dt = least(silencer_rx_dt_cols)
first_attr = ' '.join([f' when {attr_rx_dt_cols[i]} = first_treatment_rx_date then "{attr_rx_dt_cols[i].split("_")[0]}" ' for i in range(len(attr_rx_dt_cols))])
first_silencer = ' '.join([f' when {silencer_rx_dt_cols[i]} = SILENCER_first_rx_date then "{silencer_rx_dt_cols[i].split("_")[0]}" ' for i in range(len(silencer_rx_dt_cols))])
first_amv_wai = ' '.join([f' when least({",".join(wai_amv_rx_dt_cols)}) = {wai_amv_rx_dt_cols[i]} then "{wai_amv_rx_dt_cols[i].split("_")[0]}" ' for i in range(len(wai_amv_rx_dt_cols))])
first_stabilizer = ' '.join([f' when {stabilizer_rx_dt_cols[i]} = STABILIZER_first_rx_date  then "{stabilizer_rx_dt_cols[i].split("_")[0]}" ' for i in range(len(stabilizer_rx_dt_cols))])

stage2_final_view = spark.sql(f'''
select *,
case {first_attr} end as first_attr_rx,
case {first_silencer} end as first_silencer_rx,
case {first_amv_wai} end as brand_first_between_amv_wai_rx,
case {first_stabilizer} end as brand_first_stabilizer_rx
from stage2_final_view                                      
''')
stage2_final_view.createOrReplaceTempView('stage2_final_view')
