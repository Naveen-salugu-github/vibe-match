nsp_units_mnth = spark.sql(f"""
select distinct ndc_11_cd,sls_dt,sum(nsp_prch_units) as nsp_units_mnth
FROM us_commercial_app_commons_prod.f_sls_nsp_mth a
left JOIN us_commercial_app_commons_prod.d_prod b 
on a.az_prod_id=b.az_prod_id AND a.prod_lvl = b.prod_lvl
WHERE ndc_11_cd IN (select distinct ndc_cd from us_commercial_catalog_prod.lgu_eplontersen.d_rx_codes_attr) 
AND a.prod_lvl = 'CMF10' 
and a.trade_class_desc not in ('DRUG','COMBINED','PROVIDER') 
group by all
order by 3,2 """)

spark.sql("""
          select ndc_cd,mnth,
          case when ndc_cd in (select distinct ndc_cd from us_commercial_catalog_prod.lgu_patientiq.d_rx_codes_attr where prod_desc_lvl3 like '%VYNDAMAX%') then (claim_count*(day_supl/30)) -- To normalise Vyndamax and vyndaqel calims to 30 days

spark.sql("""      
select x1.ndc_cd,x1.mnth,claim_count,ddd_units,ddd_units_norm,nsp_units_mnth as nsp_units,nsp_units_norm
from common_tab as x1
left join elaad as x2
on x1.ndc_cd=x2.ndc_cd and x1.mnth=x2.mnth
left join DDD_units_norm as x3
on x1.ndc_cd=x3.ndc_cd and x1.mnth=x3.mnth
left join NSP_units_norm as x4
on x1.ndc_cd=x4.ndc_11_cd and x1.mnth=x4.sls_dt
order by mnth   
""").createOrReplaceTempView('factor_base')

## Rolling up to Brand family level and calculating the factors

============================================================
CELL 25 | code
============================================================
spark.sql('''
         select *,(claim_count/ddd_units_norm) as elaad_ddd_fctr,(claim_count/nsp_units_norm) as elaad_nsp_fctr,
(ddd_units_norm/nsp_units_norm) as ddd_nsp_fctr 
 from (
select mnth,prod_desc_lvl3 as prod_family,
 sum(claim_count) as claim_count,sum(ddd_units_norm) as ddd_units_norm,sum(nsp_units_norm) as nsp_units_norm
from factor_base as x1
inner join (select distinct ndc_cd,prod_desc_lvl3 from us_commercial_catalog_prod.lgu_patientiq.d_rx_codes_attr) as x2
on x1.ndc_cd=x2.ndc_cd
group by all
)
          ''').createOrReplaceTempView('prjc_fin')

fin_df=spark.sql('''
  select 
    mnth,
    prod_family,
    claim_count,
    ddd_units_norm,
    nsp_units_norm,
  case
    when elaad_ddd_fctr>1 then 1 else elaad_ddd_fctr
  end as elaad_ddd_fctr,
  case
    when elaad_nsp_fctr>1 then 1 else elaad_nsp_fctr
  end as elaad_nsp_fctr,
  case 
    when prod_family like "%VYNDA%" then 0.90
    when prod_family like'%AMVUTTRA%' and mnth < "2024-02-01" then 0.55 -- change as per 1/04 suggestion
    when prod_family like'%AMVUTTRA%' and mnth >= "2024-02-01" and mnth <="2025-03-01" then 0.85 -- changing to 0.85 for testing
    when prod_family like'%AMVUTTRA%' and mnth >= "2025-04-01" then 0.80 -- As per Harshit request - 31 July 2025
    when prod_family like '%ONPATTRO%' and mnth < "2024-02-01" then 0.6
    when prod_family like '%ONPATTRO%' and mnth >= "2024-02-01" then 0.85 -- changing to 0.85 for testing
    when prod_family like '%WAINUA%' then 1
  end as blocking_fctr

  from prjc_fin
  ''')
fin_df.createOrReplaceTempView('Final_NSP_projection_stg1')

                     select mnth,prod_family,claim_count,ddd_units_norm,nsp_units_norm,elaad_ddd_fctr,elaad_nsp_fctr,blocking_fctr from Final_NSP_projection
