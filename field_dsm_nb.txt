sp_base=spark.sql(f""" 
select distinct sp.*,coalesce(zip,src_cust_zip) as zip,hcp_cust_name

# Calculation of SD DU by summing up the quality shipped to SP and SD and grouping by week ending date.
sd_du = spark.sql(f"""
SELECT 
   CASE 
      WHEN dim.ndc = 00310940001 THEN "WAINUA PEN"
      WHEN dim.ndc = 00310942001 THEN "WAINUA PFS"
   END AS Product,
   dim.az_cust_id,
   coalesce(dim.src_cust_zip,cust_zip) as src_cust_zip,
   date(dim.rpt_dt) as rpt_dt,
   (case 
      when extract(dayofweek from dim.rpt_dt) =7 then DATE_ADD(dim.rpt_dt,6) 
      else DATE_ADD(dim.rpt_dt, 6 - extract(dayofweek from dim.rpt_dt)) 
   end) as wk_end_dt,
   date(date_trunc('MONTH',dim.rpt_dt)) as Month, 
   coalesce(SD.SD_DEMAND_UNITS,0) as SD_DEMAND_UNITS,
   coalesce(SD_ship.SD_SHIPMENT_UNITS_TO_SP,0) as SD_SHIPMENT_UNITS_TO_SP 
FROM
   (
      select distinct az_cust_id,
      rpt_dt,src_cust_zip,cust_zip,ndc from {sd_table}
      where mkt_nm='{mkt_nm}' and ndc in {ndc}
   ) as dim

FULL JOIN

   (select az_cust_id,rpt_dt,src_cust_zip, ndc,coalesce(sum(sum_qt),0) as SD_demand_units 
   from (
      select az_cust_id, rpt_dt, src_cust_zip, ndc,sum(ship_qty) as sum_qt 
      from {sd_table}    
      where mkt_nm='{mkt_nm}' 
      and src_dedupe_ind='N' and ndc in {ndc} group by az_cust_id,rpt_dt,src_cust_zip,ndc) 
   group by az_cust_id,rpt_dt,src_cust_zip,ndc) as SD
   on dim.rpt_dt=SD.rpt_dt and dim.az_cust_id=SD.az_cust_id and dim.src_cust_zip = SD.src_cust_zip and dim.ndc = SD.ndc

FULL JOIN

   (select az_cust_id,rpt_dt, ndc,coalesce(sum(sum_qt),0) as SD_shipment_units_to_sp 
   from (
      select az_cust_id,rpt_dt, ndc, sum(ship_qty) as sum_qt 
      from {sd_table} 
      where mkt_nm='{mkt_nm}' 
      and src_dedupe_ind='Y' and ndc in {ndc}  group by az_cust_id,rpt_dt,ndc) 
   group by az_cust_id,rpt_dt,ndc) as SD_ship
   ON dim.rpt_dt = SD_ship.rpt_dt and dim.az_cust_id=SD_ship.az_cust_id and dim.ndc = SD_ship.ndc
""")
sd_du.createOrReplaceTempView("sd_du")

spsd_districtlevel_daily = spark.sql(f'''
select
  coalesce(sd.product, sp.product) as Product,
  coalesce(sd.rpt_dt, sp.ship_dt) date,
  coalesce(sd.Month,sp.Month) Month,
  coalesce(sd.wk_end_dt,sp.wk_end_dt) week_end_dt,
  coalesce(sd.region_name,sp.region_name) Region,
  coalesce(sd.district_name,sp.district_name) District,
  sp.diag_cd,
  coalesce(sd.SD_DEMAND_UNITS,0) as SD_Demand, coalesce(sd.SD_SHIPMENT_UNITS_TO_SP, 0) as SD_Shipment_to_SP, coalesce(sp.SP_Demand_Units,0 ) as SP_Demand
from
  (
    select
      product, rpt_dt, wk_end_dt,Month,
      region_name,"null" as diag_cd,
      district_name,
      sum(SD_DEMAND_UNITS) as SD_DEMAND_UNITS,
      sum(SD_SHIPMENT_UNITS_TO_SP) as SD_SHIPMENT_UNITS_TO_SP
    from
      us_commercial_catalog_prod.lgu_eplontersen.sd_hca_idn
    group by
      all
  ) as sd full
  join (
    select
      product, ship_dt,wk_end_dt,Month,
      region_name,
      district_name,
      diag_cd,
      sum(SP_demand_units) as SP_Demand_Units
    from
      us_commercial_catalog_prod.lgu_eplontersen.sp_hcp_date
    group by
      all
  ) as sp on sd.rpt_dt = sp.ship_dt
  and sd.district_name = sp.district_name and sd.diag_cd = sp.diag_cd and sd.product = sp.product
  order by date desc, (SP_Demand_Units+SD_DEMAND_UNITS) desc                                                                     
''')
spsd_districtlevel_daily.createOrReplaceTempView("spsd_districtlevel_daily")
